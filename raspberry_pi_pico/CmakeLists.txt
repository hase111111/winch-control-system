cmake_minimum_required(VERSION 3.13)

include(pico_sdk_import.cmake)

project(pico2_udp_adc C CXX)

set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)

pico_sdk_init()

add_executable(pico2_udp_adc
    main.cpp
)

target_link_libraries(pico2_udp_adc
    pico_stdlib
    hardware_adc
    pico_cyw43_arch_lwip_threadsafe_background
)

pico_enable_stdio_usb(pico2_udp_adc 1)
pico_enable_stdio_uart(pico2_udp_adc 0)

pico_add_extra_outputs(pico2_udp_adc)
